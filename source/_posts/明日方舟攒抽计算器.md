        function parseDate(str, forceYear) {
            if (str.startsWith('N')) {
                const parts = str.slice(1).split('/').map(Number).map((v) => Math.abs(v));
                const year = forceYear - str.startsWith('N-');
                const solar = Lunar.fromYmd(year, parts[0], parts[1]).getSolar();
                return new Date(solar.getYear(), solar.getMonth() - 1, solar.getDay());
            }
            const p = str.split('/').map(Number);
            let y = forceYear;
            const now = new Date();
            const d = new Date(y, p[0]-1, p[1]);
            if (d < now && p[0] < now.getMonth()+1) y++;
            return new Date(y, p[0]-1, p[1]);
        }

        }

            const t = new Date(date); t.setHours(0,0,0,0);
            s.setHours(0,0,0,0); e.setHours(0,0,0,0);
            return t >= s && t <= e;
        }

            let totalOriginium = currentOriginium;
            let totalTickets = currentTickets;
            let dailyLog = [];

            const start = new Date(startDate);
            let current = startDate;
            let dayCount = 0;

            while (current <= endDate) {
                const dateStr = formatDate(current);
                let dayJade = 0;
                let dayOriginium = 0;
                let dayTickets = 0;
                let parts = [];

                // ---- 每日任务 ----
                if (!dailyDone) {
                    dayJade += 100;
                    parts.push('日常+100');
                }

                // ---- 月卡：从今天开始算剩余天数 ----
                const daysSinceToday = Math.floor((current - today) / (1000 * 60 * 60 * 24)) - firstDayHelpDone;
                if (!helpDone && hasMonthly && daysSinceToday >= 0 && daysSinceToday < monthlyDays) {
                    dayJade += 200;
                    parts.push('月卡+200');
                }

                // ---- 周常：只在周常刷新日发，且每周只发一次 ----
                if (current.getDay() === weekTaskDay) {
                    if (!weeklyDone) {
                        dayJade += 500;
                        parts.push('周常+500');
                    }
                    weeklyDone = false;
                }

                // ---- 剿灭：只在歼灭刷新日发，且每周只发一次 ----
                if (current.getDay() === annihilationDay) {
                    let weekAnnihilation = annihilationCap;
                    if (annihilationDay < current.getDay() && !(annihilationDay === 0)) {
                        isThisWeek = false;
                    }
                    if (isThisWeek) {
                        weekAnnihilation = Math.max(0, annihilationCap - annihilationClaimed);
                    }
                    if (weekAnnihilation > 0) {
                        dayJade += weekAnnihilation;
                        parts.push(`剿灭+${weekAnnihilation}`);
                    }
                    isThisWeek = false;
                }

                if (!activityDone) {
                    // ---- 签到 ----
                    const CheckIn = getCheckIn(current, isFirstActivity);
                    if (typeof CheckIn === 'number') { dayJade += CheckIn; parts.push(`签到+${CheckIn}`); }
                    else if (!CheckIn) isFirstActivity = false;

                    // ---- 源石签到 ----
                    if (isOriginiumDay(current, start)) { dayOriginium += 1; parts.push('源石+1'); }
                }

                // ---- 森空岛 ----
                const skland = getSkland(current, start);
                if (!noSkland && skland > 0 && !sklandDone) { dayJade += skland; parts.push(`森空岛+${skland}`); }

                // ---- 游戏内签到(17号) ----
                const gameSign = getGameSign(current, start);
                if (gameSign > 0 && !helpDone) { dayTickets += gameSign; parts.push('补给+1抽'); }

                // ---- 幸运墙 ----
                if (isWishWallDay(current) && !luckilyDone) { dayJade += luckilyJades; parts.push(`幸运墙+${luckilyJades}`); }

                // ---- 记录 ----
                if (dayJade > 0 || dayOriginium > 0 || dayTickets > 0) {
                    dailyLog.push({ date: dateStr, jade: dayJade, originium: dayOriginium, tickets: dayTickets, detail: parts.join(' · ') });
                }

                totalJade += dayJade;
                totalOriginium += dayOriginium;
                totalTickets += dayTickets;

                dailyDone = false;
                helpDone = false;
                luckilyDone = false;
                sklandDone = false;
                activityDone = false;

                dayCount++;
                current.setDate(current.getDate() + 1);
            }

            // ---- 汇总 ----
            const totalJadeFromOriginium = totalOriginium * 180;
            const totalJadeAll = totalJade + totalJadeFromOriginium;
            const totalPulls = Math.floor(totalJadeAll / 600) + totalTickets;
            const leftoverJade = totalJadeAll % 600;

            const resultBox = document.getElementById('resultBox');
            resultBox.style.display = 'block';
            document.getElementById('resultStats').innerHTML = `
                <div class="stat"><div class="num" id="result-jade">${totalJade}</div><div class="label">合成玉 (不含源石)</div></div>
                <div class="stat"><div class="num green" id="result-originium">${totalOriginium}</div><div class="label">源石</div><div class="sub">= ${totalJadeFromOriginium} 玉</div></div>
                <div class="stat"><div class="num blue" id="result-ticket">${totalTickets}</div><div class="label">寻访凭证</div></div>
                <div class="stat"><div class="num purple" id="result">${totalPulls}</div><div class="label">总抽数</div><div class="sub" id="result-more">余 ${leftoverJade} 玉</div></div>
            `;

            const logArea = document.getElementById('logArea');
            if (dailyLog.length === 0) {
                logArea.textContent = '（无任何稳定收入日期）';
                document.getElementById('downloadBtn').hidden = true;
            } else {
                logArea.textContent = dailyLog.map(e => `${e.date}  |  ${e.detail}`).join('\n');
                document.getElementById('logDaysCount').textContent = dailyLog.length;
                document.getElementById('downloadBtn').hidden = false;
            }
            resultBox.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }

        function download() {
            const logTexts = document.getElementById('logArea').textContent.split('\n');
            const infos = [
                {
                    '总抽数': document.getElementById('result').textContent,
                    '余下玉': document.getElementById('result-more').textContent.match(/\d+/g)[0],
                    '合成玉': document.getElementById('result-jade').textContent,

                    '寻访凭证': document.getElementById('result-ticket').textContent,

            let datas = []
            let maxDataLength = 0;

            for (const logText of logTexts) {
                const logTextSplit = logText.split('  |  ');
                const date = logTextSplit[0];
                const texts = logTextSplit[1];
                const logs = texts.split(' · ');

                if (logs.length > maxDataLength) maxDataLength = logs.length;
                
                let tempData = {
                    '日期': date
                };
                for (const [i, log] of logs.entries()) {
                    tempData['项目 ' + (i + 1).toString()] = log;
                }
                datas.push(tempData);

            const wb = XLSX.utils.book_new();

            const ws_data = XLSX.utils.json_to_sheet(datas);

                {wch: 15}
            ];

            for (let _ = 0; _ < maxDataLength; _++) {
                ws_data['!cols'].push({wch: 18});
            }

            XLSX.utils.book_append_sheet(wb, ws_info, '攒抽结果');
            XLSX.utils.book_append_sheet(wb, ws_data, '攒抽详情');

            XLSX.writeFile(wb, '攒抽结果.xlsx');
        }

@app.route("/login/<int:ticket>", methods=["GET"])
def login_page(ticket):
    """扫码登录确认页面

    ---
    tags:
        - 页面
    parameters:
        - name: ticket
          in: path
          type: integer
          required: true
          description: 登录票据 ID
    responses:
        200:
            description: 返回登录确认页面
        404:
            description: 票据不存在
        400:
            description: 票据已被处理
    """
    ticket_key = str(ticket)
    if ticket_key not in tickets:
        body = f"""
            <h1>错误</h1>
            <div class="info error">票据 {ticket} 不存在</div>
            <a href="{host}" class="btn-back">返回首页</a>
        """
        return render_page("错误", body), 404

    if tickets[ticket_key] != "created":
        body = f"""
            <h1>错误</h1>
            <div class="info error">票据已被处理</div>
            <a href="{host}" class="btn-back">返回首页</a>
        """
        return render_page("错误", body), 400

    tickets[ticket_key] = "scanned"
    tickets_scan[ticket_key] = True

    body = f"""
        <h1>登录确认</h1>
        <div class="info">票据 ID: {ticket}</div>
        <input type="number" id="user_id" placeholder="输入用户 ID">
        <button onclick="confirmLogin()">确认登录</button>
        <button onclick="cancelLogin()">取消</button>
        <div id="result"></div>
        <a href="{host}" class="btn-back">返回首页</a>
    """
    script = f"""
        const ticket = {ticket};
        async function confirmLogin() {{
            const userId = document.getElementById('user_id').value;
            if (!userId) {{
                document.getElementById('result').innerHTML = '<div class="info error">请输入用户 ID</div>';
                return;
            }}
            try {{
                const res = await fetch(`{host}/api/auths/${{ticket}}/confirm`, {{
                    method: 'POST',
                    headers: {{ 'Content-Type': 'application/json' }},
                    body: JSON.stringify({{ user_id: parseInt(userId) }})
                }});
                const data = await res.json();
                if (res.ok) {{
                    document.getElementById('result').innerHTML = '<div class="info success">登录确认成功</div>';
                    setTimeout(() => {{ window.location.href = '{host}'; }}, 1500);
                }} else {{
                    document.getElementById('result').innerHTML = `<div class="info error">${{escapeHtml(data.msg || '确认失败')}}</div>`;
                }}
            }} catch (err) {{
                document.getElementById('result').innerHTML = '<div class="info error">确认失败</div>';
            }}
        }}
        async function cancelLogin() {{
            try {{
                const res = await fetch(`{host}/api/auths/${{ticket}}/cancel`, {{ method: 'POST' }});
                const data = await res.json();
                if (res.ok) {{
                    document.getElementById('result').innerHTML = '<div class="info success">已取消登录</div>';
                    setTimeout(() => {{ window.location.href = '{host}'; }}, 1500);
                }} else {{
                    document.getElementById('result').innerHTML = `<div class="info error">${{escapeHtml(data.msg || '取消失败')}}</div>`;
                }}
            }} catch (err) {{
                document.getElementById('result').innerHTML = '<div class="info error">取消失败</div>';
            }}
        }}
        function escapeHtml(str) {{
            if (!str) return '';
            return str.replace(/[&<>]/g, function(m) {{
                if (m === '&') return '&amp;';
                if (m === '<') return '&lt;';
                if (m === '>') return '&gt;';
                return m;
            }});
        }}
    """
    return render_page("登录确认", body, script)

@app.put("/api/users/<string:username>")
def update_user(username):
    """更新用户信息

    修改用户名或用户简介，至少提供一个修改项

    ---
    tags:
        - 用户管理
    parameters:
        - name: username
          in: path
          type: string
          required: true
          description: 原用户名
        - name: new_username
          in: query
          type: string
          required: false
          description: 新用户名
        - name: new_description
          in: query
          type: string
          required: false
          description: 新简介
    responses:
        200:
            description: 更新成功
            schema:
                type: object
                properties:
                    username:
                        type: string
                        description: 新用户名
                        example: "lisi"
                    description:
                        type: string
                        description: 新简介
                        example: "更新后的简介"
                    user_id:
                        type: integer
                        description: 用户 ID
                        example: 1
        400:
            description: 没有提供更新内容
            schema:
                type: object
                properties:
                    status:
                        type: string
                        example: "400"
                    msg:
                        type: string
                        example: "没有提供更新内容"
        404:
            description: 用户不存在
            schema:
                type: object
                properties:
                    status:
                        type: string
                        example: "404"
                    msg:
                        type: string
                        example: "用户不存在"
        409:
            description: 新用户名已存在
            schema:
                type: object
                properties:
                    status:
                        type: string
                        example: "409"
                    msg:
                        type: string
                        example: "新用户名已存在"
    """
    if username not in usernames:
        return error_response(404, "用户不存在")

    new_username = request.args.get("new_username", "")
    new_description = request.args.get("new_description", "")

    if not new_username and not new_description:
        return error_response(400, "没有提供更新内容")

    for user_id, info in users.items():
        if info["name"] == username:
            if new_username:
                if new_username in usernames and new_username != username:
                    return error_response(409, "新用户名已存在")
                if "<" in new_username or ">" in new_username or "&" in new_username:
                    return error_response(422, "用户名包含非法字符")
                usernames.remove(username)
                usernames.append(new_username)
                info["name"] = new_username
            if new_description:
                if (
                    "<" in new_description
                    or ">" in new_description
                    or "&" in new_description
                ):
                    return error_response(422, "简介包含非法字符")
                info["desc"] = new_description
            return {
                "username": info["name"],
                "description": info["desc"],
                "user_id": int(user_id),
            }

    return error_response(404, "用户不存在")

@app.route("/qr/<path:data>", methods=["GET"])
def create_qr(data):
    """生成二维码图片

    ---
    tags:
        - 工具
    parameters:
        - name: data
          in: path
          type: string
          required: true

    responses:
        200:
            description: 返回 PNG 格式的二维码图片
            schema:
                type: file
                format: binary
    """
    qr_url = unquote(data)

    qr = qrcode.QRCode(
        version=1,
        error_correction=qrcode.constants.ERROR_CORRECT_L,
        box_size=10,
        border=4,
    )

    qr.add_data(qr_url)

    qr.make(fit=True)
    qc = qr.make_image(fill_color="black", back_color="white")

    img_buffer = io.BytesIO()
    qc.save(img_buffer, format="PNG")
    img_buffer.seek(0)

    return Response(img_buffer.getvalue(), mimetype="image/png")
